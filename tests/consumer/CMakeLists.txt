cmake_minimum_required(VERSION 3.21)
project(liboai_consumer LANGUAGES CXX)
set(LIBOAI_SOURCE_DIR "" CACHE PATH "Source checkout for add_subdirectory; empty uses installation")
option(LIBOAI_PARENT_ALIAS_TARGETS "Exercise unnamespaced vendored dependency target names" OFF)
set(LIBOAI_JSON_SOURCE_DIR "" CACHE PATH "nlohmann_json source for vendored-target check")
option(LIBOAI_PARENT_DEPENDENCIES "Supply dependency targets from parent" OFF)
if(LIBOAI_SOURCE_DIR)
  if(LIBOAI_PARENT_ALIAS_TARGETS)
    if(NOT LIBOAI_JSON_SOURCE_DIR)
      message(FATAL_ERROR "Set LIBOAI_JSON_SOURCE_DIR for the vendored dependency check")
    endif()
    add_subdirectory("${LIBOAI_JSON_SOURCE_DIR}" json-build)
    # A parent-supplied libcurl target linked to the installed platform cURL.
    # This checks liboai's target-name fallback; it does not rebuild cURL itself.
    find_library(consumer_curl NAMES curl REQUIRED)
    add_library(libcurl INTERFACE)
    target_link_libraries(libcurl INTERFACE "${consumer_curl}")
  elseif(LIBOAI_PARENT_DEPENDENCIES)
    find_package(nlohmann_json CONFIG REQUIRED)
    find_package(CURL REQUIRED)
  endif()
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(LIBOAI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(LIBOAI_INSTALL OFF CACHE BOOL "" FORCE)
  add_subdirectory("${LIBOAI_SOURCE_DIR}" liboai-build)
  set(oai_target oai)
else()
  find_package(oai CONFIG REQUIRED)
  set(oai_target oai::oai)
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE ${oai_target})
